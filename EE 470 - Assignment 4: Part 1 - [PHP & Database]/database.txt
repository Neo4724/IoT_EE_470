CREATE TABLE sensor_register (
    node_name VARCHAR(10) NOT NULL,
    manufacturer VARCHAR(10) NOT NULL,
    longitude DECIMAL(15,10) NOT NULL,
    latitude DECIMAL(15,10) NOT NULL,

    PRIMARY KEY (node_name)
);

CREATE TABLE sensor_data (
    id INT NOT NULL AUTO_INCREMENT,
    node_name VARCHAR(10) NOT NULL,
    time_received DATETIME NOT NULL,
    temperature DECIMAL(6,2) NOT NULL,
    humidity DECIMAL(6,2) NOT NULL,

    PRIMARY KEY (id),

    UNIQUE (node_name, time_received),

    CONSTRAINT fk_sensor_node
        FOREIGN KEY (node_name)
        REFERENCES sensor_register(node_name),

    CHECK (temperature >= -10 AND temperature <= 100),
    CHECK (humidity >= 0 AND humidity <= 100)
);

CREATE VIEW sensor_combined AS

SELECT
    d.node_name,
    r.manufacturer,
    r.longitude,
    r.latitude,
    d.time_received,
    d.temperature,
    d.humidity

FROM sensor_data d

INNER JOIN sensor_register r
ON d.node_name = r.node_name;
